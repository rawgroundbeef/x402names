          - 192.168.0.0/16 (192.168.0.0 - 192.168.255.255)
          - 127.0.0.0/8 (127.0.0.0 - 127.255.255.255) (loopback range)
          - 169.254.0.0/16 (169.254.0.0 - 169.254.255.255) (link-local)
          Error: "Private IP addresses are not allowed"
       f. Cloud metadata endpoints: reject hostname `169.254.169.254` or `metadata.google.internal`. Error: "Metadata service URLs are not allowed"
       g. Embedded credentials: reject if `url.username` or `url.password` is non-empty. Error: "URLs with embedded credentials are not allowed"

    2. Add `createValidationProblem` to `apps/api/src/lib/errors.ts`:

       Export a new helper function:
       ```typescript
       export function createValidationProblem(
         c: Context,
         errors: Array<{ field: string; code: string; message: string }>
       ) {
         return c.json({
           type: 'error:validation',
           title: 'Validation Failed',
           status: 400,
           detail: 'Request contains validation errors',
           errors: errors.map(e => ({
             field: e.field,
             code: e.code,
             message: e.message
           }))
         }, 400);
       }
       ```

       Also update the existing `validationErrorHook` to aggregate ALL Zod errors instead of just the first one:
       ```typescript
       export const validationErrorHook: Hook<any, any, any, any> = (result, c) => {
         if (!result.success) {
           const errors = 'errors' in result.error ? result.error.errors : [];
           const validationErrors = errors.map((err: any) => ({
             field: err.path.join('.'),
             code: 'VALIDATION_ERROR',
             message: err.message
           }));

           return c.json({
             type: 'error:validation',
             title: 'Validation Failed',
             status: 400,
             detail: 'Request contains validation errors',
             errors: validationErrors
           }, 400);
         }
       };
       ```
