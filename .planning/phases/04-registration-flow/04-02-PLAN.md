    6. **Extract payer wallet and validate payment amount from payment header**:
       The x402 payment middleware is NOT used for this route (we handle payment header directly for dynamic pricing control). The payment header contains a cryptographically signed payment authorization.

       ```typescript
       import { decodePaymentSignatureHeader } from '@x402/core/http';

       // Get payment header (x402 uses 'payment-signature' or 'x-payment')
       const paymentHeader = c.req.header('payment-signature') || c.req.header('x-payment');

       if (!paymentHeader) {
         return createProblemResponse(c, 402, 'error:payment_required', 'Payment Required', 'No payment header found');
       }

       // Decode to get PaymentPayload
       const paymentPayload = decodePaymentSignatureHeader(paymentHeader);

       // Extract payer from EVM exact scheme payload
       // The payload contains { authorization: { from: "0x..." }, signature: "0x..." }
       const payerWallet = (paymentPayload.payload as any)?.authorization?.from;

       if (!payerWallet) {
         return createProblemResponse(c, 400, 'error:invalid_payment', 'Invalid Payment', 'Could not extract payer wallet from payment');
       }
       ```

       **PAYMENT AMOUNT VALIDATION (CRITICAL):**
       Extract the payment amount from the decoded payload and validate it against the TLD price. Without this check, an agent could register a $13.18 domain by paying $0.01.

       ```typescript
       // Extract payment amount from the payload
       // The amount is in the accepted field (atomic USDC units, 6 decimals)
       // OR in the authorization value field
       const paymentAmountRaw = paymentPayload.accepted?.amount
         || (paymentPayload.payload as any)?.authorization?.value;

       if (!paymentAmountRaw) {
         return createProblemResponse(c, 400, 'error:invalid_payment', 'Invalid Payment', 'Could not extract payment amount');
       }

       // Convert atomic USDC (6 decimals) to human-readable amount
       const paymentAmountUsdc = Number(paymentAmountRaw) / 1_000_000;
       const requiredAmountUsdc = pricing!.registrationPriceUsdc;

       if (paymentAmountUsdc < requiredAmountUsdc) {
         return createProblemResponse(c, 402, 'error:insufficient_payment', 'Insufficient Payment',
           `Domain ${domain} requires ${requiredAmountUsdc.toFixed(2)} USDC but payment was ${paymentAmountUsdc.toFixed(2)} USDC`);
       }
       ```

       **SECURITY NOTE (HARD-05):**
       ```typescript
       // TODO [HARD-05]: Verify x402 payment signature server-side against the facilitator.
       // Currently we decode the payment header but do not cryptographically verify the signature.
       // This is acceptable for v1 because:
       // 1. The facilitator handles actual USDC settlement independently
       // 2. The server only needs the wallet address for domain ownership attribution
       // 3. A forged header would attribute the domain to the wrong wallet (attacker hurts themselves)
       // Phase 6 hardening will add: facilitator signature verification via x402 verify functions.
       ```

       Generate a payment ID from the payment header for idempotency:
       ```typescript
       // Use a hash of the payment header as payment ID for idempotency
       const paymentId = await generatePaymentId(paymentHeader);
       ```

       Create a helper function `generatePaymentId` that creates a SHA-256 hash of the payment header string (use `crypto.subtle.digest` or `Bun.hash`):
       ```typescript
       async function generatePaymentId(paymentHeader: string): Promise<string> {
         const hash = new Bun.CryptoHasher('sha256');
         hash.update(paymentHeader);
         return hash.digest('hex');
       }
       ```

       const jobId = crypto.randomUUID();

       // Record payment and create job in a transaction

       db.transaction(() => {
         recordPayment(db, {
           paymentId,
           walletAddress: payerWallet,
           amount: pricing!.registrationPriceUsdc.toFixed(2),
           network: c.req.header('x-payment-network') || 'unknown',
           domain,
         });

         db.insert(registrationJobs).values({
           id: jobId,
           domainName: domain,
           tld: validation.tld!,

           paymentId,
           amountPaid: pricing!.registrationPriceUsdc.toFixed(2),
           targetUrl: targetUrl || null,
           state: 'processing',
           progress: 0,
           currentStep: 'payment_verified',
           attempts: 0,
           createdAt: new Date(),
         }).run();

    ```typescript
    export function createStatusRoutes(db: BunSQLiteDatabase<any>) {
      const router = new Hono();

      router.get('/:jobId/status', (c) => {
        const jobId = c.req.param('jobId');

        const job = db
          .select()
          .from(registrationJobs)
          .where(eq(registrationJobs.id, jobId))
          .get();

        if (!job) {
          return createProblemResponse(c, 404, 'error:job_not_found', 'Job Not Found', `Registration job ${jobId} not found`);
        }

        switch (job.state) {
          case 'processing':
            return c.json({
              state: 'processing',
              progress: job.progress,
              currentStep: job.currentStep,
              retryAfterSeconds: 2,
            });

          case 'succeeded':
            return c.json({
              state: 'succeeded',
              artifactUrl: `/domains/${job.domainName}/status`,
              domain: job.domainName,
              ownerWallet: job.ownerWallet,
              response: `Domain ${job.domainName} registered successfully`,
            });

          case 'failed':
            return c.json({
              state: 'failed',
              error: job.error || 'Registration failed',
              code: job.errorCode || 'unknown_error',
              domain: job.domainName,
            });

          default:
            return createProblemResponse(c, 500, 'error:invalid_state', 'Invalid Job State', `Job has invalid state: ${job.state}`);
        }
      });

      return router;
    }
    ```
